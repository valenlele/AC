PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
PA EQU 30H
PB EQU 31H
CA EQU 32H
CB EQU 33H

ORG 40
DIR_F10 DW RUT_F10

ORG 1000H
STR DB "PRESIONE F10 PARA COMENZAR"
FIN DB ?
FLAG DB 0

ORG 2000H
CLI
MOV AL, 11111110B
OUT IMR, AL
MOV AL, 10
OUT INT0, AL

MOV AL, OFFSET FIN - OFFSET STR
MOV BX, OFFSET STR
INT 7
STI

LOOP: CMP FLAG, 1
JNZ LOOP

CALL IMPRIMIR

INT 0

ORG 3000H
RUT_F10: PUSH AX
PUSH BX
PUSH CX
PUSH DX

MOV FLAG, 1

MOV BX, SP
SUB BX, 5
MOV CL, 10

LOOP_LEER: INT 6
DEC BX
DEC CL
JNZ LOOP_LEER

MOV AL, EOI
OUT PIC, AL

POP DX
POP CX
POP BX
POP AX
IRET

STROBE0: IN AL, PA
AND AL, 11111101B
OUT PA, AL
RET

POLL: IN AL, PA
AND AL, 1
JNZ POLL
RET

STROBE1: IN AL, PA
OR AL, 00000010B
OUT PA, AL
RET

CONF_PIO: MOV AL, 0
OUT CB, AL
MOV AL, 1
OUT CA, AL
CALL STROBE0
RET 

IMPRIMIR: PUSH AX
PUSH BX
PUSH CX

CALL CONF_PIO
MOV BX, SP
SUB BX, 9
MOV CL, 10
LOOP_IMPRIMIR: CALL POLL
MOV AL, [BX]
DEC BX
OUT PB, AL
CALL STROBE1
CALL STROBE0
DEC CL
JNZ LOOP_IMPRIMIR

POP CX
POP BX
POP AX
RET

END
