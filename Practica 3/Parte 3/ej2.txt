PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
PB EQU 31H
CB EQU 33H

ORG 40
DIR_F10 DW RUT_F10

ORG 1000H
ID_F10 DB 10
FLAG DB 0

ORG 2000H
CLI
MOV AL, 11111110B
OUT IMR, AL

MOV AL, ID_F10
OUT INT0, AL

MOV AL, 0
OUT CB, AL

OUT PB, AL
STI

LOOP: JMP LOOP

ORG 3000H
RUT_F10: PUSH AX
CMP FLAG, 0
JNZ APAGAR

MOV AL, 0FFH
OUT PB, AL
MOV FLAG, 1
JMP FIN

APAGAR: MOV AL, 0
OUT PB, AL
MOV FLAG, 0

FIN: MOV AL, EOI
OUT PIC, AL

POP AX
IRET

END
