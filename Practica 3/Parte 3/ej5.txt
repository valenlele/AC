CONT EQU 10H
COMP EQU 11H
PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT1 EQU 25H
PA EQU 30H
PB EQU 31H
CA EQU 32H
CB EQU 33H

ORG 40
DIR_TIMER DW RUT_TIMER

ORG 1000H
FLAG DB 0
E_LLAVES DB ?

ORG 2000H
MOV AL, 10
OUT INT1, AL
MOV AL, 12
OUT COMP, AL
MOV AL, 0FFH
OUT CA, AL
MOV AL, 0
OUT CB, AL

LOOP: IN AL, PA
MOV E_LLAVES, AL
MOV BX, OFFSET E_LLAVES
PUSH BX
CALL LUCES12
POP BX
JMP LOOP

ORG 3000H
LUCES12: PUSH AX
PUSH BX

MOV BX, SP
ADD BX, 6
MOV AX, [BX]
MOV BX, AX
MOV AX, [BX]

AND AL, 10000000B
JZ FIN

MOV AL, 0FFH
OUT PB, AL
MOV AL, 11111101B
OUT IMR, AL
MOV AL, 0
OUT CONT, AL

LOOP_ESPERA: CMP FLAG, 1
JNZ LOOP_ESPERA

MOV AL, 11111111B
OUT IMR, AL
MOV AL, 0
OUT PB, AL
MOV FLAG, 0

FIN: POP BX
POP AX

RET

RUT_TIMER: PUSH AX

MOV FLAG, 1
MOV AL, EOI
OUT PIC, AL

POP AX
IRET

END
