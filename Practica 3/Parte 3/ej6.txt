PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
DATO EQU 40H
ESTADO EQU 41H

ORG 40
  DIR_F10 DW RUT_F10

ORG 1000H
  STR DB "INGRESE UN NUMERO DEL 1 AL 9"
  FIN DB ?
  NUM DB ?
  FLAG_F10 DB 0

ORG 2000H
  CLI
  MOV AL, 11111111B
  OUT IMR, AL
  MOV AL, 10
  OUT INT0, AL
  STI

  CALL LEER_DIGITO

  MOV AL, 11111110B
  OUT IMR, AL

  LOOP_F10: CMP FLAG_F10, 1
  JNZ LOOP_F10

  MOV AL, NUM
  MOV AH, 0
  PUSH AX  

  CALL DESCENDER

  POP AX

  INT 0

ORG 3000H
LEER_DIGITO: 
  PUSH AX
  PUSH BX
  
  MOV BX, OFFSET STR
  MOV AL, OFFSET FIN - OFFSET STR
  INT 7

  MOV BX, OFFSET NUM
  LOOP_LEER: INT 6
  MOV AL, [BX]
  CMP AL, '1'
  JS LOOP_LEER
  CMP AL, '9' + 1
  JNS LOOP_LEER

  POP BX
  POP AX
RET

RUT_F10:
  PUSH AX
  
  MOV FLAG_F10, 1
  MOV AL, EOI
  OUT PIC, AL
  
  POP AX
IRET

DESCENDER:
  PUSH AX
  PUSH BX
  PUSH CX
  
  MOV BX, SP
  ADD BX, 8
  MOV CX, [BX]

  LOOP_IMPRIMIR: MOV AL, CL
  CALL IMPRIMIR
  DEC CL
  CMP CL, '0' - 1
  JNZ LOOP_IMPRIMIR

  POP CX
  POP BX
  POP AX
RET

IMPRIMIR: 
  PUSH AX
  
  CALL POLL
  OUT DATO, AL

  POP AX
RET

POLL:
  PUSH AX
  
  LOOP_POLL: IN AL, ESTADO
  AND AL, 1
  JNZ LOOP_POLL
  
  POP AX
RET

END
