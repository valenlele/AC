PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
PA EQU 30H
PB EQU 31H
CA EQU 32H
CB EQU 33H

ORG 40
RUT_F1O DW DIR_F10

ORG 1000H
FLAG DB 0
CAR DB ?

ORG 2000H
CLI
MOV AL, 11111110B
OUT IMR, AL
MOV AL, 10
OUT INT0, AL
CALL CONF_PIO
STI

MOV CL, 0
MOV BX, OFFSET CAR

LOOP: INT 6
INC BX
INC CL
CMP FLAG, 1
JNZ LOOP

INT 0

ORG 3000H
CONF_PIO: MOV AL, 1
OUT CA, AL
MOV AL, 0
OUT CB, AL
RET

ORG 3100H
DIR_F10: PUSH AX
PUSH BX
PUSH CX

CMP CL, 0
JZ FIN
MOV FLAG, 1
MOV BX, OFFSET CAR
CALL STROBE0

LOOP_2: CALL POLL
MOV AL, [BX]
OUT PB, AL
CALL STROBE1
CALL STROBE0
INC BX
DEC CL
JNZ LOOP_2

FIN: MOV AL, EOI
OUT PIC, AL

POP CX
POP BX
POP AX
IRET

ORG 3200H
STROBE0: PUSH AX

IN AL, PA
AND AL, 11111101B
OUT PA, AL

POP AX
RET

ORG 3300H
POLL: PUSH AX
IN AL, PA
AND AL, 1
JNZ POLL

POP AX
RET

ORG 3400H
STROBE1: PUSH AX

IN AL, PA
OR AL, 00000010B
OUT PA, AL

POP AX
RET

END
