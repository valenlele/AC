CONT EQU 10H
COMP EQU 11H
PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
INT1 EQU 25H
PB EQU 31H
CB EQU 33H

ORG 40
  DIR_F10 DW RUT_F10

ORG 80
  DIR_TIMER DW RUT_TIMER

ORG 1000H
  STR DB "Hola!"
  FLAG_F10 DB 0
  FLAG_TIMER DB 0

ORG 2000H
  CLI
  MOV AL, 10
  OUT INT0, AL
  MOV AL, 20
  OUT INT1, AL
  STI

  MOV AL, 0
  OUT CB, AL

  MOV BX, OFFSET STR
  MOV CL, 5
  MOV AL, 5
  OUT COMP, AL

  MOV AL, 11111110B
  OUT IMR, AL

  LOOP:
    CMP FLAG_F10, 1
    JZ FIN_F10

    MOV AL, [BX]
    OUT PB, AL

    MOV AL, 11111100B
    OUT IMR, AL
    MOV AL, 0
    OUT CONT, AL
    LOOP_TIMER: 
      CMP FLAG_F10, 1
      JZ FIN_F10
      CMP FLAG_TIMER, 1
      JNZ LOOP_TIMER

    MOV AL, 11111110B
    OUT IMR, AL

    INC BX
    MOV FLAG_TIMER, 0
    DEC CL
    JNZ LOOP

  MOV AL, 0
  OUT PB, AL

  FIN_F10:
    INT 0

ORG 3000H
  RUT_F10:
    PUSH AX
    
    MOV FLAG_F10, 1
    MOV AL, EOI
    OUT PIC, AL

    POP AX
    IRET

  RUT_TIMER:
    PUSH AX
    
    MOV FLAG_TIMER, 1
    MOV AL, EOI
    OUT PIC, AL

    POP AX
    IRET
END
