CONT EQU 10H
COMP EQU 11H
PIC EQU 20H
EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
INT1 EQU 25H

ID_F10 EQU 10
ID_TIMER EQU 20

ORG 40
IP_F10 DW RUT_F10

ORG 80
IP_TIMER DW RUT_TIMER

ORG 1000H
VALOR DB "9"
CANT DB 10
PAUSA DB 0 ;0 = ESTOY CONTANDO ; 1 = ME INTERRUMPIERON ; 2 = TERMINE NORMAL

ORG 2000H

CLI

MOV AL, ID_F10
OUT INT0, AL

MOV AL, ID_TIMER
OUT INT1, AL

MOV AL, 0FCH
OUT IMR, AL

MOV AL, 1
OUT COMP, AL

MOV AL, 0
OUT CONT, AL

STI

LOOP: JMP LOOP

ORG 3000H
RUT_F10: PUSH AX
PUSH BX

CMP PAUSA, 2
JZ RESET

CMP PAUSA, 0
JZ INTERRUPCION

MOV PAUSA, 0
JMP FIN_3

INTERRUPCION: MOV PAUSA, 1
JMP FIN_3

RESET: MOV PAUSA, 0
MOV VALOR, '9'
MOV CANT, 10

MOV AL, 0
OUT CONT, AL

FIN_3: MOV AL, EOI
OUT PIC, AL

POP BX
POP AX
IRET

ORG 3100H
RUT_TIMER: PUSH AX
PUSH BX

CMP PAUSA, 1
JZ FIN

MOV AL, 1
MOV BX, OFFSET VALOR
INT 7

DEC BYTE PTR [BX]
DEC CANT
JNZ FIN
MOV PAUSA, 2
JMP FIN_2

FIN: MOV AL, 0
OUT CONT, AL

FIN_2: MOV AL, EOI
OUT PIC, AL

POP BX
POP AX
IRET

END
