PA EQU 30H
PB EQU 31H
CA EQU 32H
CB EQU 33H

ORG 1000H
STR_A DB "Fin de programa"
FIN_A DB ?
STR_C DB "Arquitectura de Computadoras: ACTIVADA"
FIN_C DB ?

ORG 3000H
A: MOV AH, 0
IN AL, PA
CMP AL, 0
JNZ FIN_2

MOV AH, 1
MOV BX, OFFSET STR_A
MOV AL, OFFSET FIN_A - OFFSET STR_A
INT 7

FIN_2: RET

B: IN AL, PA
NOT AL
OUT PB, AL
RET

C: IN AL, PA
AND AL, 1
JZ FIN_3
MOV BX, OFFSET STR_C
MOV AL, OFFSET FIN_C - OFFSET STR_C
INT 7

FIN_3: RET

ORG 2000H
MOV AL, 0FFH
OUT CA, AL
MOV AL, 0
OUT CB, AL

LOOP: CALL A
CMP AH, 1
JZ FIN

CALL B
CALL C
JMP LOOP

FIN: HLT

END
