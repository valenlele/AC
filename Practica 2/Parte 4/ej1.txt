PB EQU 31H
CB EQU 33H

ORG 2000H
MOV AL, 0
OUT CB, AL

LOOP: MOV AL, 0
OUT PB, AL
INC AL
OUT PB, AL

MOV CL, 7

LOOP_IZQ: MOV AH, 1
CALL ROTARIZQ_N
OUT PB, AL
DEC CL
JNZ LOOP_IZQ

MOV CL, 7

LOOP_DER: MOV AH, 1
CALL ROTARDER_N
OUT PB, AL
DEC CL
JNZ LOOP_DER

JMP LOOP

ORG 3000H
ROTARIZQ: ADD AL, AL
ADC AL, 0
RET

ROTARIZQ_N: CMP AH, 0
JZ FIN
SEGUIR: CALL ROTARIZQ
DEC AH
JNZ SEGUIR
FIN: RET

ROTARDER_N: NEG AH
ADD AH, 8
CALL ROTARIZQ_N
RET

HLT
END
