HAND EQU 40H

ORG 1000H
  X DB ?
  STR DB ?

ORG 2000H
  CALL INICIALIZAR_HANDSHAKE

  MOV BX, OFFSET X
  LEER: 
    INT 6
    SUB BYTE PTR [BX], 30H
    MOV CL, [BX]
    CMP CL, 1
    JS LEER
    CMP CL, 10
    JNS LEER
  
  MOV BX, OFFSET STR
  LOOP:
    INT 6
    INC BX
    DEC CL
    JNZ LOOP

  MOV BX, OFFSET STR
  MOV CL, X
  LOOP_IMPRIMIR: 
    CALL IMPRIMIR
    INC BX
    DEC CL
    JNZ LOOP_IMPRIMIR

  INT 0

ORG 3000H
  INICIALIZAR_HANDSHAKE:
    PUSH AX
    IN AL, HAND + 01H
    AND AL, 01111111B
    OUT HAND + 01H, AL  
    POP AX
    RET

  IMPRIMIR:
    CALL POLL
    MOV AL, [BX]
    OUT HAND, AL
    RET

  POLL:
    IN AL, HAND + 01H
    AND AL, 1
    JNZ POLL
    RET

END
